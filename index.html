IoT Mini-Game Controller

PocketParty IoT Documentation

ระบบควบคุมมินิเกมไร้สายผ่านเครือข่ายอินเทอร์เน็ต (IoT) ด้วยโปรโตคอล MQTT เชื่อมต่อผู้เล่นหลายคน รองรับอินพุตหลากหลาย ทั้งการตรวจจับความเคลื่อนไหวด้วยอัลตราโซนิก, แป้นพิมพ์ปุ่มกด, ตัวต้านทานปรับค่าได้ และจอแสดงผล LCD แสดงผลสถานะแบบ Real-time

ฮาร์ดแวร์และเซนเซอร์ที่ใช้งาน

Arduino UNO R4 WiFi

บอร์ดควบคุมหลัก ทำหน้าที่ประมวลผลเซนเซอร์ จัดการตรรกะของเกม และเชื่อมต่อเครือข่ายอินเทอร์เน็ตผ่าน Wi-Fi เพื่อสื่อสารกับ MQTT Broker

Ultrasonic Sensor (HC-SR04)

ตรวจจับระยะทางของมือผ่านคลื่นเสียง ใช้พิน Trig (10) และ Echo (11) รองรับการตรวจจับจังหวะก้าวในเกมวิ่ง (RUN) และตรวจความลึกในการปั๊มลูกโป่ง (BALLOON)

Keypad 4x4 Matrix

แป้นปุ่มกด 16 ปุ่ม (Rows: 9, 8, 7, 6 / Cols: 5, 4, 3, 2) สำหรับรับคำสั่งกดปุ่ม ตอบคำถาม หรือป้อนรหัสตัวเลขในมินิเกมต่าง ๆ

Potentiometer (A0)

ตัวต้านทานปรับค่าได้ ต่อเข้าที่ขาแอนะล็อก A0 แปลงสัญญาณแรงดันเป็นสเกล 0–100% สำหรับฟังก์ชันหมุนรหัสปลดล็อกตู้เซฟ (SAFE Mode)

LCD Display 16x2 (I2C)

จอแสดงผลขนาด 16 ตัวอักษร 2 บรรทัดผ่านบัส I2C (Address 0x27) แสดงรหัสเครื่อง (Device ID), ตำแหน่งผู้เล่น (P1/P2), โหมดเกม และข้อความแจ้งเตือน

การทำงานร่วมกับระบบ MQTT

อุปกรณ์สื่อสารผ่าน MQTT Broker โดยส่งสตรีมข้อมูลอินพุตไปยัง Topic game/thai99/stream และรอรับคำสั่งควบคุมหรือข้อความแสดงผลบนหน้าจอผ่าน Topic game/thai99/cmd ช่วยให้อุปกรณ์สลับบทบาทและรองรับโหมดเกมใหม่ได้ทันที

ซอร์สโค้ด Arduino UNO R4 WiFi

#include <WiFiS3.h>
#include <PubSubClient.h>
#include <Keypad.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

const char* ssid        = "";
const char* password    = "";
const char* mqtt_server = "broker.emqx.io";
const int   mqtt_port   = 1883;
const char* ROOM_CODE   = "thai99";

LiquidCrystal_I2C lcd(0x27, 16, 2);

const int TRIG_PIN = 10;
const int ECHO_PIN = 11;
const int POT_PIN  = A0;

const byte ROWS = 4;
const byte COLS = 4;
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {9, 8, 7, 6};
byte colPins[COLS] = {5, 4, 3, 2};
Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

WiFiClient espClient;
PubSubClient client(espClient);

char deviceId[10];
int  myRole = 0;
String currentGameMode = "ALL";

char topicStream[50];
char topicCmd[50];
char mqttClientId[30];

unsigned long lastPingTime   = 0;
unsigned long lastSensorPoll = 0;
unsigned long lastMqttSend   = 0;
const unsigned long MQTT_RATE_LIMIT = 80;

bool inStroke = false;
int  currentPistonDepth = 0;

bool handInZone = false;
unsigned long lastStepTime = 0;
const unsigned long STEP_COOLDOWN = 100;

int lastPotVal = -1;

void showLCD(String line1, String line2) {
  lcd.setCursor(0, 0);
  lcd.print(line1 + "                ");
  lcd.setCursor(0, 1);
  lcd.print(line2 + "                ");
}

void updateDefaultScreen(String statusMsg = "READY") {
  if (myRole == 0) {
    showLCD("ID: " + String(deviceId), "WAITING PAIR...");
  } else {
    showLCD("P" + String(myRole) + " [" + String(deviceId) + "] " + currentGameMode, "> " + statusMsg);
  }
}

void sendPacket(String jsonPayload, bool force = false) {
  if (!force && (millis() - lastMqttSend < MQTT_RATE_LIMIT)) return;
  lastMqttSend = millis();

  client.publish(topicStream, jsonPayload.c_str());
  Serial.println("[TX] " + jsonPayload);
}

void sendHandshakePing() {
  String payload = "{\"type\":\"PING\",\"id\":\"" + String(deviceId) + "\"}";
  sendPacket(payload, true);
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String msg = "";
  for (unsigned int i = 0; i < length; i++) msg += (char)payload[i];
  Serial.println("[RX] " + msg);

  if (msg.indexOf("WHO_IS_HERE") >= 0) {
    sendHandshakePing();
  }
  else if (msg.indexOf("\"type\":\"ROLE\"") >= 0) {
    if (msg.indexOf("\"p1\":\"" + String(deviceId) + "\"") >= 0) {
      myRole = 1;
      updateDefaultScreen("ASSIGNED P1");
    } else if (msg.indexOf("\"p2\":\"" + String(deviceId) + "\"") >= 0) {
      myRole = 2;
      updateDefaultScreen("ASSIGNED P2");
    }
  }
  else if (msg.indexOf("\"type\":\"MODE\"") >= 0) {
    int idx = msg.indexOf("\"mode\":\"");
    if (idx >= 0) {
      currentGameMode = msg.substring(idx + 8, msg.indexOf("\"", idx + 8));
      Serial.println("[MODE] Switched to: " + currentGameMode);
      updateDefaultScreen("MODE: " + currentGameMode);
    }
  }
  else if (msg.indexOf("\"type\":\"LCD\"") >= 0) {
    bool isForMe = (msg.indexOf("\"id\":\"ALL\"") >= 0) || 
                   (msg.indexOf("\"id\":\"" + String(deviceId) + "\"") >= 0) ||
                   (myRole == 1 && msg.indexOf("\"id\":\"P1\"") >= 0) ||
                   (myRole == 2 && msg.indexOf("\"id\":\"P2\"") >= 0);

    if (isForMe) {
      int idxL1 = msg.indexOf("\"l1\":\"");
      int idxL2 = msg.indexOf("\"l2\":\"");
      if (idxL1 >= 0 && idxL2 >= 0) {
        String l1 = msg.substring(idxL1 + 6, msg.indexOf("\"", idxL1 + 6));
        String l2 = msg.substring(idxL2 + 6, msg.indexOf("\"", idxL2 + 6));
        showLCD(l1, l2);
      }
    }
  }
}

long readUltrasonicCm() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 18000);
  if (duration == 0) return 999;
  return duration * 0.034 / 2;
}

void pollSensors() {
  // --- Keypad ---
  if (currentGameMode == "KEYPAD" || currentGameMode == "BALLOON" || currentGameMode == "SAFE" || currentGameMode == "ALL") {
    char key = keypad.getKey();
    if (key) {
      String payload = "{\"type\":\"KEY\",\"id\":\"" + String(deviceId) + "\",\"key\":\"" + String(key) + "\"}";
      sendPacket(payload, true);
    }
  }

  // --- Potentiometer ---
  if (currentGameMode == "SAFE" || currentGameMode == "ALL") {
    int raw = analogRead(POT_PIN);
    int potVal = constrain(map(raw, 0, 920, 0, 100), 0, 100);

    if (abs(potVal - lastPotVal) >= 2) {
      lastPotVal = potVal;
      String payload = "{\"type\":\"POT\",\"id\":\"" + String(deviceId) + "\",\"val\":" + String(potVal) + "}";
      sendPacket(payload);
    }
  }

  // --- Ultrasonic ---
  if (currentGameMode == "RUN" || currentGameMode == "BALLOON" || currentGameMode == "ALL") {
    if (millis() - lastSensorPoll >= 30) {
      lastSensorPoll = millis();
      long dist = readUltrasonicCm();
      if (currentGameMode == "RUN" || currentGameMode == "ALL") {
        if (dist >= 3 && dist <= 20) {
          handInZone = true;
        } else if (handInZone && (dist > 24 || dist == 999)) {
          handInZone = false;
          if (millis() - lastStepTime > STEP_COOLDOWN) {
            lastStepTime = millis();
            String payload = "{\"type\":\"STEP\",\"id\":\"" + String(deviceId) + "\"}";
            sendPacket(payload, true);
          }
        }
      }

      if (currentGameMode == "BALLOON" || currentGameMode == "ALL") {
        if (dist >= 4 && dist <= 25) {
          inStroke = true;
          int depth = constrain(map(dist, 25, 4, 0, 100), 0, 100);
          if (depth > currentPistonDepth) currentPistonDepth = depth;
        } else if (inStroke && (dist > 26 || dist == 999)) {
          inStroke = false;
          if (currentPistonDepth >= 10) {
            String payload = "{\"type\":\"PUMP\",\"id\":\"" + String(deviceId) + "\",\"depth\":" + String(currentPistonDepth) + "}";
            sendPacket(payload, true);
          }
          currentPistonDepth = 0;
        }
      }
    }
  }
}

void setupWiFi() {
  showLCD("Connecting WiFi", ssid);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(400);
}

void reconnectMQTT() {
  while (!client.connected()) {
    showLCD("Connecting MQTT", "Broker EMQX...");
    if (client.connect(mqttClientId)) {
      client.subscribe(topicCmd);
      updateDefaultScreen();
      sendHandshakePing();
    } else {
      delay(2000);
    }
  }
}

void setup() {
  Serial.begin(115200);

  lcd.init();
  lcd.backlight();
  showLCD("Controller Boot", "Init Sensors...");

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  keypad.setDebounceTime(80);

  randomSeed(analogRead(A1) + millis());
  sprintf(deviceId, "D%ld", random(1000, 9999));

  sprintf(topicStream, "game/%s/stream", ROOM_CODE);
  sprintf(topicCmd, "game/%s/cmd", ROOM_CODE);
  sprintf(mqttClientId, "UnoR4_%s_%lu", deviceId, millis());

  setupWiFi();
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(mqttCallback);
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) setupWiFi();
  if (!client.connected()) reconnectMQTT();
  client.loop();

  pollSensors();

  if (myRole == 0 && (millis() - lastPingTime > 3000)) {
    lastPingTime = millis();
    sendHandshakePing();
  }
}
